type User struct {
	Username string 

	SourceKey []byte

	PublicKey  userlib.PublicKeyType //用户公钥
	PrivateKey userlib.PrivateKeyType //用户私钥

	DSSignKey   userlib.DSSignKey //数字签名私钥
	DSVerifyKey userlib.DSVerifyKey //数字签名公钥
}


func InitUser(username string, password string) (userdataptr *User, err error) {
	var userdata User

	if len(username) == 0 {
		return nil, errors.New("InitUser Error: username should not be empty")
	}
	userdata.Username = username

// 将用户名进行hash后作为生成uuid的一个切片
	HashUserName := userlib.Hash([]byte(username)) 
	uuid, err := uuid.FromBytes(HashUserName[:16])
// 生成的uuid是存储用户数据呃key,因此根据生成的uuid作为key，查看该用户的数据是否已经存在
	_, exits := userlib.DatastoreGet(uuid)

	if exits {
		return nil, errors.New("InitUser Error: username has been exits")
	}
// 给新用户一个随机向量作为SourceKey
	userdata.SourceKey = userlib.RandomBytes(16)
// 给用户生成对应的公钥和私钥
	userdata.PublicKey, userdata.PrivateKey, err = userlib.PKEKeyGen()
// 给用户生成对应数字签名的公钥和私钥
	userdata.DSSignKey, userdata.DSVerifyKey, err = userlib.DSKeyGen()

	if err != nil {
		return nil, err
	}
// 将用户的公钥和私钥以及数字签名公钥和私钥储存起来
	userlib.KeystoreSet(username+"_PK", userdata.PublicKey)
	userlib.KeystoreSet(username+"_Sign", userdata.DSVerifyKey)
// 加密用户数据
// 现将用户数据转化为对应的json数据
// 利用argon2key算法，将username做为盐值从密码中派生出16字节的密钥
// 生成一个随机向量
// 利用派生出的密钥和随机向量对用户数据进行加密
	UserData, err := json.Marshal(userdata)
	EncKey := userlib.Argon2Key([]byte(password), []byte(username), 16)
	IV := userlib.RandomBytes(16)
	EncUserData := userlib.SymEnc(EncKey, IV, UserData)
// 利用argon2key算法,将username+password作为盐值从密码中派生出16字节的密钥
// 利用hmac算法，将派生出的密钥作为Key，对加密后的数据进行hash运算，该值不可逆
	HashKey := userlib.Argon2Key([]byte(password), []byte(username+password), 16)
	HashUser, err := userlib.HMACEval(HashKey, EncUserData)
// 将加密后的用户的hash值拼接在加密后的用户数据的前边，用来验证数据的完整性
	data := append(EncUserData, HashUser...)
	userlib.DatastoreSet(uuid, data)

	return &userdata, err
}

func GetUser(username string, password string) (userdataptr *User, err error) {
	var userdata User
	userdataptr = &userdata
//利用username生成的hash值生成对应的uuid
	HashUsername := userlib.Hash([]byte(username))
	uuid, err := uuid.FromBytes(HashUsername[:16])
//利用该uuid去查找对应的用户数据
	data, exits := userlib.DatastoreGet(uuid)

	if !exits {
		return nil, errors.New("GetUser Error: user is not exits")
	}
//如果用户数据长度小于64,说明该数据有误，因为在初始化数据的时候，拼接在加密后用户数据后的加密后的数据的hash值长度为64
	if len(data) < 64 {
		return nil, errors.New("GetUser Error: userData length < 64")
	}
// 利用argon2key函数，将username+password作为盐值，从password中派生出密钥，将该密钥作为参数利用hmac算法重新计算加密的用户数据的hash值是否和之前存储的一样来确认用户数据的完整性
	HashKey := userlib.Argon2Key([]byte(password), []byte(username+password), 16)
	CurrentHashUser, err := userlib.HMACEval(HashKey, data[:len(data)-64])
	if !userlib.HMACEqual(CurrentHashUser, data[len(data)-64:]) {
		return nil, errors.New("GetUser Error: User data has no integrity")
	}
// 利用argon2key函数，将username作为盐值，从password中派生出对应的密钥，对加密的用户数据进行解密
	EncKey := userlib.Argon2Key([]byte(password), []byte(username), 16)
	UserData := userlib.SymDec(EncKey, data[:len(data)-64])
// 将用户数据填充到对应的数据结构中
	err = json.Unmarshal(UserData, &userdata)

	return &userdata, err
}
